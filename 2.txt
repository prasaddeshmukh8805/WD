    <script>
        function validate(){

            let name = document.myform.name.value;
            let email = document.myform.email.value;
            let mobile = document.myform.mobile.value;

            if(name==""){
                alert("Enter Name");
                return false;
            }

            if(email==""){
                alert("Enter Email");
                return false;
            }

            if(mobile==""){
                alert("Enter Mobile Number");
                return false;
            }

            if(mobile.length != 10){
                alert("Mobile number must be 10 digits");
                return false;
            }

        }
    </script>
