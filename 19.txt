        function showPlace()
        {
            let place = document.getElementById("place").value;

            let details = "";

            if(place == "Taj Mahal")
            {
                details = "Taj Mahal is located in Agra and is one of the Seven Wonders of the World.";
            }

            else if(place == "India Gate")
            {
                details = "India Gate is a famous war memorial located in New Delhi.";
            }

            else if(place == "Gateway of India")
            {
                details = "Gateway of India is a popular tourist place located in Mumbai.";
            }

            else if(place == "Red Fort")
            {
                details = "Red Fort is a historical fort located in Delhi.";
            }

            else if(place == "Charminar")
            {
                details = "Charminar is a famous monument located in Hyderabad.";
            }

            document.getElementById("result").innerHTML =
            "<h3>Tourist Place Information</h3>" +
            "<p><b>Selected Place :</b> " + place + "</p>" +
            "<p>" + details + "</p>";
        }
