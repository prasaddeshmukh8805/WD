    <script>

        function clock()
        {
            let d = new Date();

            let h = d.getHours();

            let m = d.getMinutes();

            let s = d.getSeconds();

            let t = h + ":" + m + ":" + s;

            let dt = d.toDateString();

            document.getElementById("time").innerHTML = t;

            document.getElementById("date").innerHTML = dt;
        }

        setInterval(clock, 1000);

        clock();

    </script>
