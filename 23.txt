    <script>

        function clock()
        {
            var d=new Date();

            var h=d.getHours();

            var m=d.getMinutes();

            var s=d.getSeconds();

            var t=h+":"+m+":"+s;

            document.getElementById("time").innerHTML=t;
        }

        setInterval(clock,1000);

    </script>
