    <script>

        function calculateFine()
        {
            var days=document.getElementById("days").value;

            var type=document.getElementById("booktype").value;

            var fine=0;

            switch(type)
            {
                case "Story":
                    fine=days*2;
                    break;

                case "Programming":
                    fine=days*5;
                    break;

                case "Science":
                    fine=days*3;
                    break;

                default:
                    fine=0;
            }

            document.getElementById("result").innerHTML=
            "Total Fine = Rs. " + fine;
        }

    </script>
